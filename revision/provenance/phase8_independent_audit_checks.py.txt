def read_column(file, group, key):
    node = file[group][key]
    if isinstance(node, h5py.Group):
        categories = node['categories'].asstr()[:]
        codes = node['codes'][:]
        return np.asarray([categories[code] if code >= 0 else '' for code in codes])
    return node.asstr()[:] if node.dtype.kind in 'OSU' else node[:]


def original_recipe(gene, tfs):
    """Independent direct transcription of historical code's sampling sequence."""
    rng = np.random.RandomState(42)
    indices = np.where(gene == 'non-targeting')[0]
    if len(indices) > 400:
        indices = rng.choice(indices, 400, replace=False)
    unique, counts = np.unique(gene, return_counts=True)
    present = {name: int(count) for name, count in zip(unique, counts)
               if name != 'non-targeting' and count >= 15}
    factors = sorted([name for name in present if name.upper() in tfs], key=lambda name: -present[name])[:48]
    nonfactors = list(rng.permutation(sorted([name for name in present if name.upper() not in tfs])))[:12]
    groups = [('control', indices.tolist())]
    for target in factors + nonfactors:
        indices = np.where(gene == target)[0]
        if len(indices) > 25:
            indices = rng.choice(indices, 25, replace=False)
        groups.append((str(target), indices.tolist()))
    return groups

report = {'source_sha256': {str(path.relative_to(PROJECT)): hashlib.sha256(path.read_bytes()).hexdigest()

with h5py.File(WORK/'data/ReplogleWeissman2022_K562_essential.h5ad') as file:
    gene = read_column(file, 'obs', 'gene')
    guide = read_column(file, 'obs', 'guide_id')
    batch = read_column(file, 'obs', 'batch')
    groups = select_groups(gene, set(targets))
    independent = original_recipe(gene, set(targets))
    assert [(group['group'], group['cells']) for group in groups] == independent
    old = json.loads((PROJECT.parent/'runs/2b_atlas/phase8v2_L16.json').read_text())['targets']
    assert [row['target'] for row in old] == [group['group'] for group in groups[1:]]
    assert [row['n_cells'] for row in old] == [len(group['cells']) for group in groups[1:]]
    flattened = [cell for group in groups for cell in group['cells']]
    assert len(flattened) == len(set(flattened))
    assert all(np.all(gene[group['cells']] == ('non-targeting' if group['kind']=='control' else group['group']))
               for group in groups)
    tf_groups = [group for group in groups if group['kind']=='tf']
    report['designs']['k562_original48'] = {
        'n_tfs': len(tf_groups), 'n_nontfs': sum(group['kind']=='nontf' for group in groups),
        'n_controls': len(groups[0]['cells']), 'n_cells_total': len(flattened),
        'exact_rng_recipe_agreement': True, 'historical_target_order_and_count_agreement': True,
        'original_cell_ids_previously_saved': False, 'groups_disjoint': True,
        'observed_labels_match_selected_groups': True,
        'tf_groups_with_self_TRRUST_edge': [group['group'] for group in tf_groups if group['group'].upper() in targets[group['group'].upper()]],
        'guide_counts_per_tf': {group['group']: len(set(guide[group['cells']])) for group in tf_groups},
        'batch_counts_per_tf': {group['group']: len(set(batch[group['cells']])) for group in tf_groups},
        'control_guide_count': len(set(guide[groups[0]['cells']])),
        'control_batch_count': len(set(batch[groups[0]['cells']])),
        'n_tf_groups_with_single_guide':sum(len(set(guide[group['cells']]))==1 for group in tf_groups),
        'n_tf_groups_with_two_guides':sum(len(set(guide[group['cells']]))==2 for group in tf_groups),
        'guide_count_source_column':'obs/guide_id', 'batch_count_source_column':'obs/batch',
        'perturbation_assignment_source_column':'obs/gene'}

# Illustrative same-gene alignment for deletion indices r<rc and the appended 513th gene.
base = [f'G{i}' for i in range(512)]
long = base + ['G512']
for tf_rank, control_rank in [(0, 1), (39, 43), (505, 509)]:

    control_deleted = long[:control_rank] + long[control_rank+1:]
    for original in range(control_rank+1, 512):
        assert base[original] == tf_deleted[original-1] == control_deleted[original-1]
